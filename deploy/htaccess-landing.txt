# iD+ by Mhawer — landing page (Vite SPA) production .htaccess
# LiteSpeed-compatible (LiteSpeed reads Apache directives natively).

# ---------------------------------------------------------------------------
# SPA fallback: route all non-file, non-directory requests to /index.html
# ---------------------------------------------------------------------------
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteBase /

  # IMPORTANT: the Laravel API lives in the SAME docroot at /ID-platform/.
  # Deep API paths like /ID-platform/api/v1/... are not real files, so the SPA
  # fallback below would rewrite them to /index.html and break the API. Pass
  # anything under /ID-platform/ straight through to its own routing.
  RewriteRule ^ID-platform(/|$) - [L]

  # Serve existing files and directories as-is
  RewriteCond %{REQUEST_FILENAME} -f [OR]
  RewriteCond %{REQUEST_FILENAME} -d
  RewriteRule ^ - [L]

  # Everything else falls back to the SPA entry point — except requests that
  # look like static assets: a missing asset must 404, not return index.html.
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteCond %{REQUEST_URI} !\.(js|css|map|json|svg|png|jpe?g|webp|gif|ico|woff2?|ttf)$ [NC]
  RewriteRule ^ /index.html [L]
</IfModule>

# ---------------------------------------------------------------------------
# Caching
#   - /assets/* is content-hashed by Vite  -> cache 1 year, immutable
#   - other statics (noise.svg, /images/*, favicon.svg) are NOT hashed
#     -> cache 1 day, no immutable
#   - index.html must never be cached so new deploys are picked up
# Scoping is done with SetEnvIf on the request URI (no <If> blocks, works on
# Apache 2.2/2.4 and LiteSpeed).
# ---------------------------------------------------------------------------
<IfModule mod_setenvif.c>
  # Non-hashed static files anywhere on the site
  SetEnvIf Request_URI "\.(js|mjs|css|woff2?|ttf|otf|eot|svg|png|jpe?g|gif|webp|avif|ico|map|json)$" CACHE_STATIC=1
  # Content-hashed Vite bundles live under /assets/ (overrides CACHE_STATIC below)
  SetEnvIf Request_URI "^/assets/" CACHE_HASHED=1
</IfModule>

<IfModule mod_headers.c>
  # Non-hashed statics: 1 day (declared first so the /assets/ rule wins)
  Header set Cache-Control "public, max-age=86400" env=CACHE_STATIC

  # Hashed build assets under /assets/: 1 year, immutable
  Header set Cache-Control "public, max-age=31536000, immutable" env=CACHE_HASHED

  # HTML entry point: always revalidate
  <FilesMatch "^index\.html$">
    Header set Cache-Control "no-cache, no-store, must-revalidate"
    Header set Pragma "no-cache"
    Header set Expires "0"
  </FilesMatch>
</IfModule>

# Disable mod_expires so inherited/type-based Expires rules cannot override
# the scoped Cache-Control headers above.
<IfModule mod_expires.c>
  ExpiresActive Off
</IfModule>

# ---------------------------------------------------------------------------
# Compression (mod_deflate; LiteSpeed honors these directives)
# ---------------------------------------------------------------------------
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html
  AddOutputFilterByType DEFLATE text/plain
  AddOutputFilterByType DEFLATE text/css
  AddOutputFilterByType DEFLATE text/javascript
  AddOutputFilterByType DEFLATE application/javascript
  AddOutputFilterByType DEFLATE application/x-javascript
  AddOutputFilterByType DEFLATE application/json
  AddOutputFilterByType DEFLATE image/svg+xml
  AddOutputFilterByType DEFLATE font/woff2
</IfModule>

# ---------------------------------------------------------------------------
# Misc hardening
# ---------------------------------------------------------------------------
Options -Indexes
